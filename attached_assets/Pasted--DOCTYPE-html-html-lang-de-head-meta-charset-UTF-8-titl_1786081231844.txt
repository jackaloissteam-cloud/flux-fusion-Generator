    function generatePrompt() {
      const motif = document.getElementById('motif').value;
      const a = document.getElementById('artistA').value;
      const b = document.getElementById('artistB').value;
      const fusion = document.getElementById('fusion').value;
      const comp = document.getElementById('composition').value;
      const light = document.getElementById('lighting').value;
      const palette = document.getElementById('palette').value;
      const detail = document.getElementById('detail').value;

      const prompt =
        motif + " im Stil von " + a + " + " + b + ", " +
        "Fusion: " + fusion + ", " +
        "Komposition: " + comp + ", " +
        "Licht: " + light + ", " +
        "Farbwelt: " + palette + ", " +
        "Detailgrad: " + detail + ", " +
        "Rendering: hochauflösend, sauber, klar.";

      document.getElementById('output').value = prompt;
    }
